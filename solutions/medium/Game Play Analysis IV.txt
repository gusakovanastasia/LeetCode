// Title: Game Play Analysis IV
            // Difficulty: Medium
            // Language: PostgreSQL
            // Link: https://leetcode.com/problems/game-play-analysis-iv/

SELECT
    ROUND(SUM(CASE WHEN a2.event_date IS NOT NULL THEN 1 ELSE 0 END) * 1.0 / COUNT(DISTINCT a1.player_id), 2) AS fraction
FROM
    (SELECT 
        player_id,
        min(event_date) AS event_date
    FROM
        Activity
    GROUP BY 
        player_id
    ) a1
LEFT JOIN
    Activity a2
    ON a1.player_id = a2.player_id
    AND a1.event_date + 1  = a2.event_date
